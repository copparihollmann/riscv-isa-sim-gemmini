cmake_minimum_required(VERSION 3.20.0)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(iris_compiler_zephyr_pipeline)
if(NOT IRIS_KERNEL_DIR OR NOT IRIS_TARGET_DIR OR NOT CONFIG_RISCV_IRIS_VME_CONTEXT)
  message(FATAL_ERROR "Requires generated OOT kernel, target setup hook and derived context kernel")
endif()
zephyr_compile_options(-fno-tree-vectorize -fno-tree-slp-vectorize)
target_include_directories(app PRIVATE ${IRIS_KERNEL_DIR} ${IRIS_TARGET_DIR}/zephyr)
set_property(SOURCE src/main.c APPEND PROPERTY COMPILE_OPTIONS -std=gnu99)
target_sources(app PRIVATE src/main.c)
if(IRIS_MLIR_MODEL_OBJECT OR IRIS_MLIR_BRIDGE_OBJECT)
  if(NOT IRIS_MLIR_MODEL_OBJECT OR NOT IRIS_MLIR_BRIDGE_OBJECT)
    message(FATAL_ERROR "MLIR mode requires both model and descriptor bridge objects")
  endif()
  target_sources(app PRIVATE ${IRIS_MLIR_MODEL_OBJECT} ${IRIS_MLIR_BRIDGE_OBJECT})
endif()
if(CONFIG_IRIS_PIPELINE_SCALAR_TWIN)
  target_compile_definitions(app PRIVATE MERLIN_MATRIX_SCALAR_TILE MERLIN_MATRIX_TILE_EDGE=32)
endif()
