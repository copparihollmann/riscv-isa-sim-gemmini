cmake_minimum_required(VERSION 3.20.0)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(iris_mixed9_firstuse_diagnostic_v2)
target_sources(app PRIVATE src/main.c src/probe.c)
target_include_directories(app PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../zephyr)
zephyr_compile_options(-fno-tree-vectorize -fno-tree-slp-vectorize)
if(CONFIG_IRIS_BENCH_VME)
  # The pinned Saturn header uses the GNU `asm` keyword.
  set_property(SOURCE src/main.c APPEND PROPERTY COMPILE_OPTIONS -std=gnu99)
  if(NOT CONFIG_RISCV_IRIS_VME_CONTEXT)
    message(FATAL_ERROR "VME requires the derived Iris Zephyr ownership/context patch")
  endif()
  if(NOT SATURN_ROOT)
    message(FATAL_ERROR "VME requires -DSATURN_ROOT=<pinned saturn checkout> and verified vector/tile context support")
  endif()
  target_include_directories(app PRIVATE ${SATURN_ROOT}/benchmarks/common)
endif()
