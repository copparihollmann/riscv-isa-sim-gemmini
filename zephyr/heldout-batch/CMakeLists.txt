cmake_minimum_required(VERSION 3.20.0)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(iris_frozen_heldout_batch)
if(NOT IRIS_KERNEL_DIR OR NOT IRIS_TARGET_DIR OR NOT CONFIG_RISCV_IRIS_VME_CONTEXT)
  message(FATAL_ERROR "Requires frozen generated kernels and explicit VME-context kernel")
endif()
zephyr_compile_options(-fno-tree-vectorize -fno-tree-slp-vectorize)
target_include_directories(app PRIVATE ${IRIS_KERNEL_DIR} ${IRIS_TARGET_DIR}/zephyr)
set_property(SOURCE src/main.c APPEND PROPERTY COMPILE_OPTIONS -std=gnu99)
target_sources(app PRIVATE src/main.c)
foreach(op gemm conv2d)
  foreach(kind shipped twin)
    set(stem batch_${op}_${kind})
    target_sources(app PRIVATE ${IRIS_KERNEL_DIR}/${stem}.c ${IRIS_KERNEL_DIR}/${stem}_model.o ${IRIS_KERNEL_DIR}/${stem}_bridge.o)
    set_property(SOURCE ${IRIS_KERNEL_DIR}/${stem}.c APPEND PROPERTY COMPILE_OPTIONS -std=gnu99)
    set_property(SOURCE ${IRIS_KERNEL_DIR}/${stem}.c APPEND PROPERTY COMPILE_DEFINITIONS merlin_vme_gemm_i8=${stem}_inner)
    if(kind STREQUAL "twin")
      set_property(SOURCE ${IRIS_KERNEL_DIR}/${stem}.c APPEND PROPERTY COMPILE_DEFINITIONS MERLIN_MATRIX_SCALAR_TILE MERLIN_MATRIX_TILE_EDGE=32)
    endif()
  endforeach()
endforeach()
