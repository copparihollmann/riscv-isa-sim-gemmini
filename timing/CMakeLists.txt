cmake_minimum_required(VERSION 3.16)
project(iris_timing LANGUAGES CXX)
add_library(iris_timing iris_timing.cpp)
target_compile_features(iris_timing PUBLIC cxx_std_17)
target_include_directories(iris_timing PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(iris_timing PRIVATE -Wall -Wextra -Werror)
add_executable(iris_timing_test test_timing.cpp)
target_link_libraries(iris_timing_test PRIVATE iris_timing)
target_compile_options(iris_timing_test PRIVATE -Wall -Wextra -Werror)
enable_testing()
add_test(NAME online_multicore COMMAND iris_timing_test)
add_library(iris_memory_path memory_path.cpp)
target_link_libraries(iris_memory_path PUBLIC iris_timing)
target_compile_options(iris_memory_path PRIVATE -Wall -Wextra -Werror)
add_executable(iris_memory_path_test test_memory_path.cpp)
target_link_libraries(iris_memory_path_test PRIVATE iris_memory_path)
target_compile_options(iris_memory_path_test PRIVATE -Wall -Wextra -Werror)
add_test(NAME shared_memory_paths COMMAND iris_memory_path_test)
add_library(iris_scalar_memory scalar_memory.cpp)
target_link_libraries(iris_scalar_memory PUBLIC iris_memory_path)
target_compile_options(iris_scalar_memory PRIVATE -Wall -Wextra -Werror)
add_executable(iris_scalar_memory_test test_scalar_memory.cpp)
target_link_libraries(iris_scalar_memory_test PRIVATE iris_scalar_memory)
target_compile_options(iris_scalar_memory_test PRIVATE -Wall -Wextra -Werror)
add_test(NAME scalar_memory_routing COMMAND iris_scalar_memory_test)
